-- PLAYER EVENTS
--============================================================

track(Players.PlayerAdded:Connect(function(player)
    State.FriendCache[player.UserId] = nil

    track(player.CharacterAdded:Connect(function()
        task.wait(0.35)

        if CFG.FriendsESP and isRobloxFriend(player) then
            createFriendVisual(player)
        end

        if tpPage.Visible then
            refreshTPList()
        end
    end))

    if tpPage.Visible then
        refreshTPList()
    end

    if CFG.FriendsESP then
        task.spawn(function()
            if isRobloxFriend(player) then
                createFriendVisual(player)
            end
        end)
    end
end))

track(Players.PlayerRemoving:Connect(function(player)
    removeFriendVisual(player)
    State.FriendCache[player.UserId] = nil

    if State.TPSpamTarget == player then
        stopTPSpam("jogador saiu")
    end

    if State.TPFollowTarget == player then
        stopTPFollow("jogador saiu")
    end

    if State.SelectedTarget == player then
        State.SelectedTarget = nil
        tpPopup.Visible = false
    end

    if tpPage.Visible then
        task.defer(refreshTPList)
    end
end))

for _, player in ipairs(Players:GetPlayers()) do
    if player ~= LP then
        track(player.CharacterAdded:Connect(function()
            task.wait(0.35)

            if CFG.FriendsESP and isRobloxFriend(player) then
                createFriendVisual(player)
            end
        end))
    end
end

track(LP.CharacterAdded:Connect(function(char)
    stopTPSpam("respawn")
    stopTPFollow("respawn")

    State.Character = char
    State.Humanoid = char:WaitForChild("Humanoid")
    State.Root = char:WaitForChild("HumanoidRootPart")

    State.OriginalWalkSpeed = State.Humanoid.WalkSpeed
    State.OriginalAutoRotate = State.Humanoid.AutoRotate

    State.GhostOriginal = setmetatable({}, {__mode = "k"})

    destroyFlyObjects()
    destroySpinObjects()

    task.wait(0.25)

    if CFG.Ghost then
        applyGhost()
    end

    if CFG.Fly then
        ensureFlyObjects()
    end

    if CFG.Spin then
        ensureSpinObjects()
    end

    if CFG.FriendsESP then
        refreshFriendESP()
    end
end))

--============================================================
-- MAIN LOOP
--============================================================

track(RunService.Heartbeat:Connect(function(dt)
    if not State.Alive then
        return
    end

    if CFG.SpeedEnabled then
        updateSpeed()
    end

    if CFG.Ghost then
        applyGhost()
    end

    if CFG.Fly then
        updateFly()
    end

    if State.VerticalDirection ~= 0 then
        updateVerticalMovement()
    end

    if CFG.Spin then
        updateSpin()
    end

    if State.TPSpamActive then
        updateTPSpam(dt)
    end

    if State.TPFollowMode then
        updateTPFollow()
    end
end))

--============================================================
-- CLEANUP
--============================================================

function Controller:Destroy()
    if not State.Alive then
        return
    end

    State.Alive = false

    stopTPSpam()
    stopTPFollow()

    CFG.Fly = false
    CFG.Spin = false
    CFG.Ghost = false
    CFG.SpeedEnabled = false
    CFG.FriendsESP = false

    restoreGhost()
    restoreSpeed()
    destroyFlyObjects()
    stopSpin()

    for player in pairs(State.FriendVisuals) do
        removeFriendVisual(player)
    end

    for _, conn in ipairs(State.Connections) do
        pcall(function()
            conn:Disconnect()
        end)
    end

    if State.Gui then
        pcall(function()
            State.Gui:Destroy()
        end)
    end
end

track(close.MouseButton1Click:Connect(function()
    Controller:Destroy()
end))

if getgenv then
    getgenv().__MOBILE_UTILITY_MENU_V1 = Controller
end

status.Text = "Menu carregado"

return Controller
