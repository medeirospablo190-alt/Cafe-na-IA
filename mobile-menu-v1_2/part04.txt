    return true
end

local function startTPSpam(player)
    local targetRoot = getPlayerRoot(player)
    if not targetRoot then
        return false
    end

    stopTPFollow()
    stopTPSpam()

    State.TPSpamActive = true
    State.TPSpamTarget = player
    State.TPSpamAngle = 0
    State.TPSpamStartedAt = os.clock()
    State.SuppressTPSpamCancelUntil = os.clock() + 0.45

    setTPSpamNoclip()

    if State.StatusLabel then
        State.StatusLabel.Text = "TP Spam: " .. player.Name
    end

    return true
end

local function updateTPSpam(dt)
    if not State.TPSpamActive then
        return
    end

    local target = State.TPSpamTarget
    local targetRoot = getPlayerRoot(target)

    if not targetRoot then
        stopTPSpam("alvo indisponível")
        return
    end

    local _, hum, root = ensureCharacter()

    -- Movimento manual cancela.
    if os.clock() > State.SuppressTPSpamCancelUntil then
        if hum.MoveDirection.Magnitude > 0.08 then
            stopTPSpam("você se moveu")
            return
        end

        local state = hum:GetState()
        if hum.Jump == true or state == Enum.HumanoidStateType.Jumping then
            stopTPSpam("você pulou")
            return
        end
    end

    setTPSpamNoclip()

    State.TPSpamAngle += dt * CFG.TPSpamSpeed

    local radius = CFG.TPSpamRadius
    local offset = Vector3.new(
        math.cos(State.TPSpamAngle) * radius,
        0.5,
        math.sin(State.TPSpamAngle) * radius
    )

    local position = targetRoot.Position + offset
    root.CFrame = CFrame.lookAt(position, targetRoot.Position)
    root.AssemblyLinearVelocity = Vector3.zero
end
