local function track(conn)
    table.insert(State.Connections, conn)
    return conn
end

local function clamp(v, a, b)
    return math.max(a, math.min(b, v))
end

local function getCharacter()
    local char = LP.Character or LP.CharacterAdded:Wait()
    local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart") or char:WaitForChild("HumanoidRootPart")

    State.Character = char
    State.Humanoid = hum
    State.Root = root

    if State.OriginalWalkSpeed == nil then
        State.OriginalWalkSpeed = hum.WalkSpeed
    end

    if State.OriginalAutoRotate == nil then
        State.OriginalAutoRotate = hum.AutoRotate
    end

    return char, hum, root
end

local function ensureCharacter()
    if not State.Character
        or not State.Character.Parent
        or not State.Humanoid
        or not State.Humanoid.Parent
        or not State.Root
        or not State.Root.Parent then
        getCharacter()
    end
    return State.Character, State.Humanoid, State.Root
end

getCharacter()

local function new(className, props, parent)
    local obj = Instance.new(className)
    for k, v in pairs(props or {}) do
        obj[k] = v
    end
    obj.Parent = parent
    return obj
end

local function addCorner(obj, radius)
    return new("UICorner", {
        CornerRadius = UDim.new(0, radius or 8)
    }, obj)
end

local function addStroke(obj, transparency)
    return new("UIStroke", {
        Thickness = 1,
        Transparency = transparency or 0.65,
        Color = Color3.fromRGB(85, 85, 85),
    }, obj)
end

local function setButtonState(button, enabled, label)
    button.Text = label .. ": " .. (enabled and "ON" or "OFF")
    button.BackgroundColor3 = enabled
        and Color3.fromRGB(29, 72, 43)
        or Color3.fromRGB(31, 31, 31)
end

local function applyGhostPart(part)
    if not part:IsA("BasePart") then
        return
    end

    if State.GhostOriginal[part] == nil then
        State.GhostOriginal[part] = part.CanCollide
    end

    part.CanCollide = false
end

local function applyGhost()
    local char = ensureCharacter()

    if not CFG.Ghost then
        return
    end

    for _, obj in ipairs(char:GetDescendants()) do
        if obj:IsA("BasePart") then
            applyGhostPart(obj)
        end
    end
end

local function restoreGhost()
    for part, old in pairs(State.GhostOriginal) do
        if part and part.Parent then
            pcall(function()
                part.CanCollide = CFG.Ghost and false or old
            end)
        end
    end

    State.GhostOriginal = setmetatable({}, {__mode = "k"})
end
