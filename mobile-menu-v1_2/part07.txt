        or "TP SPAM: OFF"

    tpFrontButton.Text =

        and "SEMPRE NA FRENTE: ON"
        or "SEMPRE NA FRENTE: OFF"

local function makePlayerRow(player)
    if player == LP then
        return
    end

    local row = new("Frame", {
        Size = UDim2.new(1, -4, 0, 42),
        BackgroundColor3 = Color3.fromRGB(24, 24, 24),
        BorderSizePixel = 0,
    }, tpPage)
    addCorner(row, 8)

    table.insert(playerRows, row)

    new("TextLabel", {
        Size = UDim2.new(1, -126, 1, 0),
        Position = UDim2.fromOffset(9, 0),
        BackgroundTransparency = 1,
        Text = player.DisplayName .. "  @" .. player.Name,
        TextColor3 = Color3.fromRGB(235, 235, 235),
        Font = Enum.Font.GothamSemibold,
        TextSize = 10,
        TextXAlignment = Enum.TextXAlignment.Left,
        TextTruncate = Enum.TextTruncate.AtEnd,
    }, row)

    local tpButton = new("TextButton", {
        Size = UDim2.fromOffset(50, 30),
        Position = UDim2.new(1, -112, 0, 6),
        BackgroundColor3 = Color3.fromRGB(38, 38, 38),
        BorderSizePixel = 0,
        Text = "TP",
        TextColor3 = Color3.new(1, 1, 1),
        Font = Enum.Font.GothamBold,
        TextSize = 11,
    }, row)
    addCorner(tpButton, 7)

    local tpPlusButton = new("TextButton", {
        Size = UDim2.fromOffset(50, 30),
        Position = UDim2.new(1, -56, 0, 6),
        BackgroundColor3 = Color3.fromRGB(38, 38, 38),
        BorderSizePixel = 0,
        Text = "TP+",
        TextColor3 = Color3.new(1, 1, 1),
        Font = Enum.Font.GothamBold,
        TextSize = 11,
    }, row)
    addCorner(tpPlusButton, 7)

    track(tpButton.MouseButton1Click:Connect(function()
        stopTPSpam()

        if teleportToPlayer(player) then
            status.Text = "TP → " .. player.Name
        else
            status.Text = "Jogador indisponível"
        end
    end))

    track(tpPlusButton.MouseButton1Click:Connect(function()
        openTPPlus(player)
    end))
end

local function refreshTPList()
    clearTPRows()

    local query = string.lower(State.TPSearchText or "")

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP then
            local username = string.lower(player.Name or "")
            local displayName = string.lower(player.DisplayName or "")

            if query == ""
                or string.find(username, query, 1, true)
                or string.find(displayName, query, 1, true) then
                makePlayerRow(player)
            end
        end
    end
end

track(tpSearchBox:GetPropertyChangedSignal("Text"):Connect(function()
    State.TPSearchText = tpSearchBox.Text or ""
    refreshTPList()
end))

refreshTPList()
